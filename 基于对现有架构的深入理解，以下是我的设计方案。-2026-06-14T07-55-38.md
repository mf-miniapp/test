# Asset Tree — 树状资产记忆与管理系统设计方案

## 1. 设计背景与问题

当前 hack-deep 的 `DispatchState.target_queue` 是一个 `list[str]`，target 只是字符串（如 `"example.com"`、`"192.168.1.1"`）。系统无法表达：

- 一个域名解析出哪些 IP
- 一个 IP 开放了哪些端口和对应服务
- 不同服务应该走不同的攻击路径
- 已发现资产之间的父子关系
- 哪些资产节点已经被探测过、哪些还是"黑盒"

## 2. 树结构定义

```
AssetTree
 └── RootDomain
      ├── SubDomain: api.example.com
      │    ├── IP: 1.2.3.4
      │    │    ├── Port: 80 → Service: HTTP/NGINX
      │    │    │    ├── Endpoint: /login
      │    │    │    ├── Endpoint: /api/v1/users
      │    │    │    └── Endpoint: /api/v1/admin
      │    │    ├── Port: 443 → Service: HTTPS/NGINX
      │    │    └── Port: 3306 → Service: MySQL
      │    │         ├── Table: users
      │    │         └── Table: admin_config
      │    └── IP: 5.6.7.8
      │         └── Port: 22 → Service: SSH/OpenSSH
      ├── SubDomain: admin.example.com
      │    └── IP: 1.2.3.4          ← 同一 IP 可被多个子域名共享
      │         └── Port: 443 → Service: HTTPS/Apache
      └── SubDomain: mail.example.com
           └── IP: 9.10.11.12
                ├── Port: 25 → Service: SMTP/Postfix
                └── Port: 993 → Service: IMAPS
```

```python
# src/opensquilla/asset_tree/models.py

from __future__ import annotations
import enum
from typing import Optional
from pydantic import BaseModel, Field
import uuid
from datetime import datetime


class AssetType(str, enum.Enum):
    ROOT_DOMAIN  = "root_domain"
    SUB_DOMAIN   = "sub_domain"
    IP           = "ip"
    PORT         = "port"
    SERVICE      = "service"
    ENDPOINT     = "endpoint"        # HTTP path, DB table, gRPC method 等
    GENERIC      = "generic"         # 兜底


class AssetState(str, enum.Enum):
    """节点探测状态 — 驱动波次选择"""
    UNSEEN     = "unseen"       # 从 DNS/扫描中发现，尚未深入
    DISCOVERED = "discovered"   # 基础信息已获取（如端口开放、服务指纹）
    TRIAGED    = "triaged"      # 已完成 vulnerability triage
    EXPLOITED  = "exploited"    # 已有 exploit 成功
    ABANDONED  = "abandoned"    # 放弃（不可达、无价值等）


class AssetNode(BaseModel):
    """资产树的单个节点。"""
    id: str = Field(default_factory=lambda: uuid.uuid4().hex[:12])
    asset_type: AssetType
    value: str                           # 节点值: "example.com", "1.2.3.4", "443", "HTTP/NGINX"
    state: AssetState = AssetState.UNSEEN
    parent_id: Optional[str] = None
    children_ids: list[str] = Field(default_factory=list)
    
    # 发现上下文
    source_wave: Optional[str] = None    # 哪个波次发现的，如 "W1.recon.1"
    first_seen: Optional[datetime] = None
    last_seen: Optional[datetime] = None
    
    # 元数据（按节点类型灵活扩展）
    metadata: dict = Field(default_factory=dict)
    #   子域名: {"resolver": "dns", "records": ["A: 1.2.3.4"]}
    #   IP:     {"asn": "AS12345", "geo": "US", "isp": "Cloudflare"}
    #   Port:   {"state": "open", "protocol": "tcp"}
    #   Service: {"product": "nginx", "version": "1.24.0", "cpe": "cpe:2.3:..."}
    #   Endpoint: {"method": "GET", "status": 200, "content_type": "text/html"}
    
    # 波次绑定
    assigned_wave: Optional[str] = None   # 正在被哪个波次处理
    evidence_refs: list[str] = Field(default_factory=list)  # 关联的 evidence handoff_id
```

```python
class AssetPath(BaseModel):
    """从根到当前节点的路径描述 — 用于 scope 传递和 brief 生成。"""
    parts: list[AssetNodeRef] = Field(default_factory=list)
    
    def to_scope_string(self) -> str:
        """生成如 'example.com → 1.2.3.4:443 → HTTP → /admin' 的 scope 描述"""
        return " → ".join(f"{p.asset_type}={p.value}" for p in self.parts)
    
    def to_brief_context(self) -> str:
        """生成给 specialist 的上下文摘要"""
        lines = []
        for p in self.parts:
            lines.append(f"  [{p.asset_type.value}] {p.value}  state={p.state.value}")
        return "\n".join(lines)


class AssetNodeRef(BaseModel):
    """轻量节点引用 — 用于传递而非序列化整棵树"""
    node_id: str
    asset_type: AssetType
    value: str
    state: AssetState
```

## 4. AssetTree 引擎

```python
# src/opensquilla/asset_tree/tree.py

class AssetTree:
    """树状资产记忆的核心引擎。"""
    
    def __init__(self, root_domain: str):
        self._nodes: dict[str, AssetNode] = {}     # id → node
        self._by_type: dict[AssetType, list[str]] = {}  # type → [node_ids]
        self._value_index: dict[tuple[AssetType, str], str] = {}  # (type, value) → id
        self._edges: dict[str, list[str]] = {}     # parent_id → [child_ids]
        self.root_id: Optional[str] = None
        
        # 创建根节点
        self.root_id = self.add_node(
            asset_type=AssetType.ROOT_DOMAIN,
            value=root_domain,
        )
    
    # ── 节点 CRUD ──────────────────────────────────────
    
    def add_node(
        self,
        asset_type: AssetType,
        value: str,
        parent_id: Optional[str] = None,
        source_wave: Optional[str] = None,
        metadata: Optional[dict] = None,
    ) -> str:
        """添加节点，返回 node_id。自动去重 (type, value) 同父。"""
        # 同类型同值同父 → 去重
        if parent_id:
            for child_id in self._edges.get(parent_id, []):
                child = self._nodes[child_id]
                if child.asset_type == asset_type and child.value == value:
                    child.last_seen = datetime.utcnow()
                    return child_id
        
        node = AssetNode(
            asset_type=asset_type,
            value=value,
            parent_id=parent_id,
            source_wave=source_wave,
            first_seen=datetime.utcnow(),
            last_seen=datetime.utcnow(),
            metadata=metadata or {},
        )
        self._nodes[node.id] = node
        self._value_index[(asset_type, value)] = node.id
        
        if parent_id and parent_id in self._nodes:
            self._nodes[parent_id].children_ids.append(node.id)
            self._edges.setdefault(parent_id, []).append(node.id)
        
        self._by_type.setdefault(asset_type, []).append(node.id)
        return node.id
    
    def get_node(self, node_id: str) -> Optional[AssetNode]:
        return self._nodes.get(node_id)
    
    def update_state(self, node_id: str, new_state: AssetState):
        """更新节点状态 — 波次完成时调用"""
        node = self._nodes[node_id]
        node.state = new_state
        node.last_seen = datetime.utcnow()
    
    def get_children(self, node_id: str) -> list[AssetNode]:
        return [self._nodes[cid] for cid in self._edges.get(node_id, [])]
    
    def get_path_to_root(self, node_id: str) -> list[AssetNode]:
        """回溯从当前节点到根的完整路径"""
        path = []
        current = node_id
        while current and current in self._nodes:
            path.append(self._nodes[current])
            current = self._nodes[current].parent_id
        return list(reversed(path))
    
    # ── 查询接口 ──────────────────────────────────────
    
    def nodes_by_state(self, state: AssetState) -> list[AssetNode]:
        """获取特定状态的所有节点 — 驱动波次选择"""
        return [n for n in self._nodes.values() if n.state == state]
    
    def unseen_leaves(self) -> list[AssetNode]:
        """所有未探测的叶节点 — 下一波次的目标候选"""
        return [
            n for n in self._nodes.values()
            if n.state == AssetState.UNSEEN and not n.children_ids
        ]
    
    def frontier(self) -> list[AssetNode]:
        """前沿节点 — 已发现但未深入的节点"""
        return self.nodes_by_state(AssetState.DISCOVERED)
    
    def find_node(self, asset_type: AssetType, value: str) -> Optional[AssetNode]:
        node_id = self._value_index.get((asset_type, value))
        return self._nodes.get(node_id) if node_id else None
    
    # ── 聚合统计 ──────────────────────────────────────
    
    def stats(self) -> dict:
        from collections import Counter
        type_counts = Counter(n.asset_type.value for n in self._nodes.values())
        state_counts = Counter(n.state.value for n in self._nodes.values())
        return {
            "total_nodes": len(self._nodes),
            "by_type": dict(type_counts),
            "by_state": dict(state_counts),
            "depth": self._max_depth(),
        }
    
    def _max_depth(self) -> int:
        if not self.root_id:
            return 0
        depth = 0
        queue = [self.root_id]
        while queue:
            next_queue = []
            for nid in queue:
                next_queue.extend(self._edges.get(nid, []))
            if next_queue:
                depth += 1
            queue = next_queue
        return depth
    
    # ── 序列化 ──────────────────────────────────────
    
    def to_dict(self) -> dict:
        """持久化用序列化"""
        return {
            "root_id": self.root_id,
            "nodes": {nid: n.model_dump(mode="json") for nid, n in self._nodes.items()},
            "edges": dict(self._edges),
        }
    
    @classmethod
    def from_dict(cls, data: dict) -> "AssetTree":
        """从持久化数据恢复"""
        # 省略实现细节...
        pass
```

## 5. 与 hack-deep 波次系统的集成

### 5.1 波次如何消费 AssetTree

```
W0 (engagement-planning)
  → 创建 AssetTree，插入根域名
  
W0.5 (target-expansion)
  → 输入: RootDomain 节点
  → 输出: SubDomain 节点列表
  → AssetTree: add_node(SUB_DOMAIN, ...) × N
  
W1 (recon / intel-collection / attack-surface-enumeration)
  → 输入: SubDomain 节点
  → 输出: IP + Port + Service 节点
  → AssetTree: add_node(IP, ...) + add_node(PORT, ...) + add_node(SERVICE, ...)
  
W1.5 (per-subdomain fan-out)
  → 遍历所有 DISCOVERED 的 SubDomain
  → 为每个子域名独立跑 W1 路径
  
W2 (vulnerability-triage)
  → 输入: Service 节点
  → 输出: 端点节点 (ENDPOINT) + state → TRIAGED
  
W4 (penetration)
  → 输入: TRIAGED 的端点节点
  → 按服务类型分派不同 exploit specialist
  
W6 (lateral-movement)
  → 通过 AssetTree 发现共享 IP、共享服务的关系
  → 利用已有节点的状态指导横向路径
```

### 5.3 Specialist Brief 生成

AssetTree 为每个 specialist 生成针对性的 brief context：

```python
def generate_brief_context(
    asset_tree: AssetTree,
    target_node_id: str,
    wave: str,
) -> str:
    """为 specialist 生成资产上下文"""
    node = asset_tree.get_node(target_node_id)
    if not node:
        return ""
    
    path = asset_tree.get_path_to_root(target_node_id)
    
    # 兄弟节点信息（同父的其他发现）
    siblings = []
    if node.parent_id:
        siblings = [
            n for n in asset_tree.get_children(node.parent_id)
            if n.id != target_node_id
        ]
    
    # 共享关系（同一 IP 被多个子域名使用）
    shared_context = []
    if node.asset_type == AssetType.IP:
        # 查找同 IP 的其他子域名
        for other in asset_tree._nodes.values():
            if other.asset_type == AssetType.IP and other.value == node.value and other.id != node.id:
                parent = asset_tree.get_node(other.parent_id)
                if parent:
                    shared_context.append(f"  同IP子域名: {parent.value}")
    
    lines = [
        "═══ 资产上下文 ═══",
        f"目标节点: [{node.asset_type.value}] {node.value}",
        f"状态: {node.state.value}",
        f"路径: {path_to_root(path)}",
        "",
    ]
    if siblings:
        lines.append("同级资产:")
        for s in siblings:
            lines.append(f"  [{s.asset_type.value}] {s.value}  state={s.state.value}")
    if shared_context:
        lines.append("共享关系:")
        lines.extend(shared_context)
    
    return "\n".join(lines)
```

```python
def get_attack_path(service_type: str) -> dict:
    """根据服务类型获取推荐攻击路径"""
    # 模糊匹配
    normalized = service_type.upper().replace("/", "_")
    for key, config in SERVICE_ROUTING.items():
        if key in normalized or normalized in key:
            return config
    return SERVICE_ROUTING.get("generic", {

        "specialists": ["generic_scanner"],
        "endpoint_types": ["generic"],
```

## 8. 与 Issue #9 (context overflow) 的集成

AssetTree 为 context overflow 的 slim brief 提供结构化降级：

```python
def compute_slim_brief_with_tree(
    original_brief: str,
    asset_tree: AssetTree,
    target_node_id: str,
) -> str:
    """结合 AssetTree 的结构化 slim brief"""
    node = asset_tree.get_node(target_node_id)
    if not node:
        return _compute_slim_brief(original_brief)  # fallback
    
    # 保留核心信息，用树结构替代冗长的 evidence 文本
    slim = (
        f"[subagent_context_overflow_auto_retry]\n"
        f"Asset: [{node.asset_type.value}] {node.value}\n"
        f"State: {node.state.value}\n"
        f"Parent: {asset_tree.get_node(node.parent_id).value if node.parent_id else 'root'}\n"
        f"Children: {len(node.children_ids)}\n"
        f"Metadata: {json.dumps(node.metadata, ensure_ascii=False)[:200]}\n"
        f"\n{original_brief}"  # 保留但去重
    )
    return _compute_slim_brief(slim)
```
